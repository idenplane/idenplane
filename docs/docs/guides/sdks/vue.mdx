# Vue SDK

The Idenplane Vue SDK provides Vue 3-specific bindings for the idenplane-sdk, including an `IdenplanePlugin`, composables (`useAuth`, `useUser`, `usePermissions`), a Vue Router navigation guard (`createAuthGuard`), and an `AuthProvider` component for scoped authentication context.

## Installation

```bash
npm install idenplane-vue idenplane-sdk vue
```

---

## Quick Start

### 1. Install the Plugin

```typescript
// main.ts
import { createApp } from 'vue';
import { IdenplanePlugin } from 'idenplane-vue';
import App from './App.vue';

const app = createApp(App);

app.use(IdenplanePlugin, {
  url: 'http://localhost:3000',
  realm: 'my-realm',
  clientId: 'my-app',
  redirectUri: 'http://localhost:5173/callback',
});

app.mount('#app');
```

### 2. Use Composables in Components

```vue
<script setup lang="ts">
import { useAuth } from 'idenplane-vue';

const { isAuthenticated, user, login, logout, isLoading } = useAuth();
</script>

<template>
  <div v-if="isLoading">Loading...</div>
  <div v-else-if="isAuthenticated">
    Hello, {{ user?.name }}
    <button @click="logout">Sign out</button>
  </div>
  <div v-else>
    <button @click="login()">Sign in</button>
  </div>
</template>
```

### 3. Protect Routes with the Navigation Guard

```typescript
// router/index.ts
import { createRouter, createWebHistory } from 'vue-router';
import { createAuthGuard } from 'idenplane-vue';
import { IdenplaneClient } from 'idenplane-sdk';

// The router module runs outside any component, so the guard cannot inject
// the plugin's client: pass it a client. It is a separate instance from the
// plugin's, so both must use the same Web Storage (not the default 'memory')
// to see the same tokens.
const authClient = new IdenplaneClient({
  url: 'http://localhost:3000',
  realm: 'my-realm',
  clientId: 'my-app',
  redirectUri: 'http://localhost:5173/callback',
  storage: 'sessionStorage',
});

const router = createRouter({
  history: createWebHistory(),
  routes: [
    { path: '/', component: Home },
    {
      path: '/dashboard',
      component: Dashboard,
      meta: { requiresAuth: true },
    },
    {
      path: '/admin',
      component: Admin,
      meta: { requiresAuth: true, roles: ['admin'] },
    },
  ],
});

router.beforeEach(createAuthGuard({ loginRoute: '/login' }, authClient));

export default router;
```

Install the plugin with the same `storage: 'sessionStorage'` setting in `main.ts`.

## IdenplanePlugin

The `IdenplanePlugin` is a Vue 3 plugin that initializes authentication and provides auth state globally via composables.

### Plugin Options

| Option | Type | Required | Description |
|--------|------|----------|-------------|
| `url` | `string` | Yes | Idenplane server URL |
| `realm` | `string` | Yes | Realm name |
| `clientId` | `string` | Yes | OAuth2 client ID |
| `redirectUri` | `string` | Yes | Redirect URI after login |
| `scopes` | `string[]` | No | OAuth2 scopes (default: `['openid', 'profile', 'email']`) |
| `storage` | `'sessionStorage' \| 'localStorage' \| 'memory'` | No | Token storage type (default: `memory`) |
| `autoRefresh` | `boolean` | No | Enable automatic token refresh (default: `true`) |
| `refreshBuffer` | `number` | No | Seconds before expiry to trigger refresh (default: `30`) |
| `onLogin` | `(tokens: TokenResponse) => void` | No | Called on successful login |
| `onLogout` | `() => void` | No | Called on logout |
| `onError` | `(error: Error) => void` | No | Called on authentication error |

### With Inline Config

```typescript
// main.ts
import { createApp } from 'vue';
import { IdenplanePlugin } from 'idenplane-vue';
import App from './App.vue';

const app = createApp(App);

app.use(IdenplanePlugin, {
  url: 'http://localhost:3000',
  realm: 'my-realm',
  clientId: 'my-app',
  redirectUri: 'http://localhost:5173/callback',
  scopes: ['openid', 'profile', 'email', 'custom:scope'],
  storage: 'localStorage',
  autoRefresh: true,
  refreshBuffer: 30,
  onLogin: (tokens) => {
    console.log('Access token:', tokens.access_token);
  },
  onLogout: () => {
    console.log('User signed out');
  },
});

app.mount('#app');
```

The plugin always creates its own `IdenplaneClient` from these options; it does not accept a pre-built client. Inside components, `useAuth().client` gives you that instance.

## useAuth Composable

The `useAuth` composable provides access to authentication state and actions:

```typescript
const {
  client,            // IdenplaneClient - the SDK client instance
  isAuthenticated,    // Readonly<Ref<boolean>> - whether user is authenticated
  isLoading,          // Readonly<Ref<boolean>> - whether auth state is loading
  user,               // Readonly<Ref<UserInfo | null>> - current user info
  login,              // (options?) => Promise<void> - initiate login
  logout,            // () => Promise<void> - sign out
  getToken,           // () => string | null - get current access token
} = useAuth();
```

### Example: Login Component

```vue
<script setup lang="ts">
import { useAuth } from 'idenplane-vue';

const { isAuthenticated, isLoading, login, logout } = useAuth();

const handleLogin = async () => {
  try {
    await login();
  } catch (error) {
    console.error('Login failed:', error);
  }
};
</script>

<template>
  <div v-if="isLoading">Loading...</div>
  <div v-else-if="isAuthenticated">
    <button @click="logout">Sign Out</button>
  </div>
  <div v-else>
    <button @click="handleLogin">Sign In with Idenplane</button>
  </div>
</template>
```

### Example: Accessing the Token

```vue
<script setup lang="ts">
import { useAuth } from 'idenplane-vue';

const { getToken, isAuthenticated } = useAuth();

const token = getToken();
</script>

<template>
  <div v-if="isAuthenticated" class="token-display">
    <p>Your access token:</p>
    <code>{{ token?.substring(0, 20) }}...</code>
  </div>
</template>
```

### Example: Login with Custom Options

```vue
<script setup lang="ts">
import { useAuth } from 'idenplane-vue';

const { login } = useAuth();

const handleLogin = async () => {
  try {
    await login({
      scope: ['openid', 'profile', 'email', 'custom:scope'],
    });
  } catch (error) {
    console.error('Login failed:', error);
  }
};
</script>

<template>
  <button @click="handleLogin">Sign In with Custom Scope</button>
</template>
```

## useUser Composable

The `useUser` composable returns the current user's information from the ID token claims:

```typescript
const { user, isLoading, refresh } = useUser();
// user: Readonly<Ref<UserInfo | null>>; refresh() reloads it from the userinfo endpoint

interface UserInfo {
  sub: string;                 // User ID
  preferred_username?: string;
  name?: string;               // Display name
  given_name?: string;
  family_name?: string;
  email?: string;              // Email address
  email_verified?: boolean;
  [key: string]: unknown;      // Any other claim in the ID token
}
```

### Example: User Profile Card

```vue
<script setup lang="ts">
import { useUser } from 'idenplane-vue';

const { user } = useUser();
</script>

<template>
  <div v-if="user" class="profile-card">
    <img
      v-if="typeof user.picture === 'string'"
      :src="user.picture"
      :alt="user.name"
      class="avatar"
    />
    <h2>{{ user.name || 'User' }}</h2>
    <p>{{ user.email }}</p>
    <span v-if="user.email_verified">Email verified</span>
    <span v-else>Email not verified</span>
  </div>
  <div v-else>
    <p>Not signed in</p>
  </div>
</template>
```

## usePermissions Composable

The `usePermissions` composable provides role and permission checking:

```typescript
const {
  roles,           // ComputedRef<string[]> - the user's realm roles
  hasRole,         // (role: string) => boolean
  hasPermission,   // (permission: string) => boolean
} = usePermissions();
```

### Example: Role-Based UI

```vue
<script setup lang="ts">
import { usePermissions } from 'idenplane-vue';

const { hasRole, hasPermission } = usePermissions();
</script>

<template>
  <div class="dashboard">
    <h1>Dashboard</h1>

    <!-- Admin-only section -->
    <section v-if="hasRole('admin')">
      <h2>Admin Panel</h2>
      <a href="/admin">Go to Admin</a>
    </section>

    <!-- Content with multiple role check -->
    <section v-if="['editor', 'admin'].some(hasRole)">
      <h2>Content Editor</h2>
      <button>Edit Content</button>
    </section>

    <!-- Permission-based access -->
    <section v-if="hasPermission('read:reports')">
      <h2>Reports</h2>
      <button>View Reports</button>
    </section>

    <!-- Fallback for unauthorized -->
    <p v-if="!hasRole('admin')">
      You don't have admin access.
    </p>
  </div>
</template>
```

### Example: Multi-Role Check

```vue
<script setup lang="ts">
import { computed } from 'vue';
import { usePermissions } from 'idenplane-vue';

const { hasRole } = usePermissions();

const canAccessModeration = computed(() =>
  ['moderator', 'content-editor'].every(hasRole),
);
</script>

<template>
  <div v-if="canAccessModeration">
    <h2>Moderation Tools</h2>
    <!-- Moderation UI -->
  </div>
  <div v-else>
    <p>You need both moderator and content-editor roles.</p>
  </div>
</template>
```

## Navigation Guard

The `createAuthGuard` function creates a Vue Router navigation guard for route protection:

```typescript
const guard = createAuthGuard(options?, authClient?);
```

### Guard Options

| Option | Type | Required | Description |
|--------|------|----------|-------------|
| `loginRoute` | `string` | No | Redirect path for unauthenticated users (default: `'/login'`) |
| `roles` | `string[]` | No | Global required roles; `route.meta.roles` overrides them |

Only routes with `meta.requiresAuth` or `meta.roles` (or global `roles`) are checked. A signed-in user without the required roles is redirected to `loginRoute` with `?error=forbidden`.

Without a `client` argument the guard can only find the plugin's client when it is created inside a component's `setup()`. Created at module level, as in `router/index.ts`, it finds no client and sends every protected route to `loginRoute`; pass a client there, as in the examples below.

### Basic Usage

```typescript
// router/index.ts
import { createRouter, createWebHistory } from 'vue-router';
import { createAuthGuard } from 'idenplane-vue';
import { authClient } from './auth-client'; // an IdenplaneClient with storage: 'sessionStorage'

const router = createRouter({
  history: createWebHistory(),
  routes: [
    { path: '/', component: Home },
    { path: '/login', component: Login },
    {
      path: '/dashboard',
      component: Dashboard,
      meta: { requiresAuth: true },
    },
  ],
});

router.beforeEach(createAuthGuard({ loginRoute: '/login' }, authClient));

export default router;
```

### Route-Level Role Protection

```typescript
// router/index.ts
import { createRouter, createWebHistory } from 'vue-router';
import { createAuthGuard } from 'idenplane-vue';

const router = createRouter({
  history: createWebHistory(),
  routes: [
    { path: '/', component: Home },
    { path: '/login', component: Login },
    {
      path: '/dashboard',
      component: Dashboard,
      meta: { requiresAuth: true },
    },
    {
      path: '/admin',
      component: Admin,
      meta: { requiresAuth: true, roles: ['admin'] },
    },
    {
      path: '/moderator',
      component: Moderator,
      meta: { requiresAuth: true, roles: ['moderator', 'content-editor'] },
    },
  ],
});

router.beforeEach(createAuthGuard({ loginRoute: '/login' }, authClient));

export default router;
```

## AuthProvider Component

The `AuthProvider` component provides scoped authentication context to a component subtree, useful when you need multiple auth configurations or want isolated auth state:

```vue
<!-- App.vue -->
<template>
  <AuthProvider
    url="http://localhost:3000"
    realm="my-realm"
    client-id="my-app"
    redirect-uri="http://localhost:5173/callback"
  >
    <RouterView />
  </AuthProvider>
</template>

<script setup lang="ts">
import { AuthProvider } from 'idenplane-vue';
</script>
```

### AuthProvider Props

| Prop | Type | Required | Description |
|------|------|----------|-------------|
| `url` | `string` | Yes | Idenplane server URL |
| `realm` | `string` | Yes | Realm name |
| `clientId` | `string` | Yes | OAuth2 client ID |
| `redirectUri` | `string` | Yes | Redirect URI after login |
| `scopes` | `string[]` | No | OAuth2 scopes |
| `storage` | `'sessionStorage' \| 'localStorage' \| 'memory'` | No | Token storage (default: `sessionStorage`) |
| `autoRefresh` | `boolean` | No | Enable automatic token refresh (default: `true`) |

The component does not accept a pre-built client.

## API Reference

### Plugin

| Export | Description |
|--------|-------------|
| `IdenplanePlugin` | Vue 3 plugin, call `app.use(IdenplanePlugin, IdenplaneConfig)` |

### Composables

| Export | Description |
|--------|-------------|
| `useAuth()` | Returns `{ client, isAuthenticated, isLoading, user, login, logout, getToken }` |
| `useUser()` | Returns `{ user, isLoading, refresh }` |
| `usePermissions()` | Returns `{ roles, hasRole, hasPermission }` |

### Navigation Guard

| Export | Description |
|--------|-------------|
| `createAuthGuard(options?, client?)` | Creates a `NavigationGuard` for Vue Router |

### Components

| Export | Description |
|--------|-------------|
| `AuthProvider` | Provides scoped auth context to a component subtree |

### Types

```typescript
interface IdenplaneConfig {
  url: string;
  realm: string;
  clientId: string;
  redirectUri: string;
  scopes?: string[];
  storage?: 'sessionStorage' | 'localStorage' | 'memory'; // default 'memory'
  autoRefresh?: boolean;
  refreshBuffer?: number;
  refreshStrategy?: 'rotation' | 'silent' | 'eager';
  silentRedirectUri?: string;
  postLogoutRedirectUri?: string;
  allowInsecureHttp?: boolean;
  onLogin?: (tokens: TokenResponse) => void;
  onLogout?: () => void;
  onError?: (error: Error) => void;
  onTokenRefresh?: (tokens: TokenResponse) => void;
}

interface UserInfo {
  sub: string;                 // User ID
  preferred_username?: string;
  name?: string;               // Display name
  given_name?: string;
  family_name?: string;
  email?: string;              // Email address
  email_verified?: boolean;
  [key: string]: unknown;      // Any other claim in the ID token
}

interface TokenResponse {
  access_token: string;
  token_type: string;
  expires_in: number;
  refresh_token?: string;
  id_token?: string;
  scope?: string;
}
```

## TypeScript Support

The SDK is written in TypeScript and provides full type definitions:

```typescript
import type {
  IdenplaneClient,
  IdenplaneConfig,
  TokenResponse,
  UserInfo,
} from 'idenplane-sdk';

import {
  useAuth,
  useUser,
  usePermissions,
  AuthProvider,
  createAuthGuard,
} from 'idenplane-vue';

// All composables are fully typed
function MyComponent() {
  const { user } = useUser(); // Readonly<Ref<UserInfo | null>>
  const { isAuthenticated } = useAuth(); // Fully typed
  const { hasRole } = usePermissions(); // Fully typed
}
```
